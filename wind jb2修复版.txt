--[[
  wind JB2 修复版
  修复内容：
    1. 恢复完整 UI 库 URL（此前被截断为 ".../UI.lua.txt" 导致 UI 加载失败）
    2. 修正两个服务器按钮多余的 end 导致的语法错误
    3. 所有子脚本按钮统一加 pcall 保护，避免个别 URL 失效时脚本中断
    4. request 兼容 http_request
    5. 移除重复的 Window:OnClose 与未使用的死代码
]]

local PATRIOT_RAW_URL = "https://raw.githubusercontent.com/jxndjdnjsnz/wind-JB/refs/heads/main/PatriotUi.luau.txt"
local SCRIPT_TITLE = "wind JB"
local DISCORD_LINK = "https://discord.gg/9178"
local GET_KEY_LINK = ""

local success, Patriot = pcall(function()
    return loadstring(game:HttpGet(PATRIOT_RAW_URL))()
end)
if not success or not Patriot then
    warn("Patriot 库加载失败！检查你的raw链接是否有效")
    return
end

Patriot.Appearance.Title = SCRIPT_TITLE
Patriot.Appearance.Subtitle = "请输入密钥以继续使用脚本"
Patriot.Links.Discord = DISCORD_LINK
Patriot.Links.GetKey = GET_KEY_LINK
Patriot.Theme.Accent = Color3.fromRGB(139, 0, 0)

Patriot.Options.Blur = true
Patriot.Options.Draggable = true
Patriot.Options.Keyless = false
Patriot.Options.KeylessUI = true

Patriot.Storage.Remember = true
Patriot.Storage.AutoLoad = true

Patriot.Callbacks.OnSuccess = function()
    -- UI 库多源自动切换：raw 失败自动切 jsdelivr CDN
    local UI_URLS = {
        "https://raw.githubusercontent.com/jxndjdnjsnz/wind-JB/refs/heads/main/UI.lua.txt",
        "https://cdn.jsdelivr.net/gh/jxndjdnjsnz/wind-JB@main/UI.lua.txt",
    }
    local WindUI
    for _, url in ipairs(UI_URLS) do
        local okFetch, src = pcall(function() return game:HttpGet(url) end)
        if okFetch and type(src) == "string" and #src > 1000 then
            local okLoad, fn = pcall(loadstring, src)
            if okLoad and fn then
                local okExec, mod = pcall(fn)
                if okExec and type(mod) == "table" and mod.CreateWindow then
                    WindUI = mod
                    break
                end
            end
        end
        warn("[JB Hub] 尝试源失败: " .. url)
    end
    if not WindUI then
        warn("[JB Hub] UI 库所有源均加载失败，请检查网络或稍后重试")
        return
    end

    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local RunService = game:GetService("RunService")
    local VirtualUser = game:GetService("VirtualUser")
    local localPlayer = Players.LocalPlayer

    local antiAfkRunning = false
    local antiAfkConnection = nil

    -- 兼容 request / http_request
    local httpRequest = request or http_request

    local Window = WindUI:CreateWindow({
        Title = "JB Hub",
        Size = UDim2.fromOffset(440, 340),
        Theme = "Dark"
    })

    -- 强制显示主窗口（手机端默认隐藏，需要强制弹出）
    task.wait(0.5)
    pcall(function() Window:Open() end)
    pcall(function() Window:Visible(true) end)
    -- 直接操作 ScreenGui 强制显示
    pcall(function()
        local players = game:GetService("Players")
        local playerGui = players.LocalPlayer:WaitForChild("PlayerGui")
        local coreGui = game:GetService("CoreGui")
        local screenGui = coreGui:FindFirstChild("WindUI") or playerGui:FindFirstChild("WindUI")
        if screenGui then
            screenGui.Enabled = true
            for _, v in pairs(screenGui:GetDescendants()) do
                if v:IsA("GuiObject") then
                    v.Visible = true
                    v.Active = true
                end
            end
        end
    end)
    pcall(function() WindUI:Notify({Title="JB Hub", Content="脚本加载成功", Icon="check"}) end)

    local Tab = Window:Tab({ Title = "服务器", Icon = "server" })
    local ContyTab = Window:Tab({ Title = "其他脚本", Icon = "box" })
    local TabOther = Window:Tab({ Title = "其他工具", Icon = "wrench" })

    Tab:Section({ Title = "请选择服务器" })
    Tab:Button({
        Title = "被遗弃",
        Callback = function()
            pcall(function()
                loadstring(game:HttpGet("https://raw.githubusercontent.com/jxndjdnjsnz/wind-JB/refs/heads/main/wind%20jb%20.txt"))()
            end)
        end
    })
    Tab:Button({
        Title = "UBG 终极战场",
        Callback = function()
            pcall(function()
                loadstring(game:HttpGet("https://raw.githubusercontent.com/jxndjdnjsnz/wind-JB/refs/heads/main/wind%20JB%20ubg.txt"))()
            end)
        end
    })

    ContyTab:Section({ Title = "其他脚本" })
    ContyTab:Button({
        Title = "RB脚本付费版",
        Callback = function()
            pcall(function()
                loadstring(game:HttpGet("https://raw.githubusercontent.com/jxndjdnjsnz/wind-JB/refs/heads/main/Rb%E8%84%9A%E6%9C%AC%E4%B8%AD%E5%BF%83%E6%BA%90%E7%A0%81.lua"))()
            end)
        end
    })
    ContyTab:Button({
        Title = "xi pro",
        Callback = function()
            pcall(function()
                if httpRequest then
                    loadstring(httpRequest({ Url = "https://raw.githubusercontent.com/Zer0neK/SB-Xi-pro/refs/heads/main/SBXiPro.lua" }).Body)()
                else
                    loadstring(game:HttpGet("https://raw.githubusercontent.com/Zer0neK/SB-Xi-pro/refs/heads/main/SBXiPro.lua"))()
                end
            end)
        end
    })
    ContyTab:Button({
        Title = "XK旧版",
        Callback = function()
            pcall(function()
                loadstring(game:HttpGet(('https://github.com/devslopo/DVES/raw/main/XK%20Hub')))()
            end)
        end
    })
    ContyTab:Button({
        Title = "XKRemake",
        Callback = function()
            pcall(function()
                loadstring(game:HttpGet("https://raw.githubusercontent.com/SyndromeXph/XK-Script/refs/heads/main/XoneK-Loader.luau"))()
            end)
        end
    })
    ContyTab:Button({
        Title = "XK测试版",
        Callback = function()
            pcall(function()
                loadstring(game:HttpGet("https://api.luarmor.net/files/v4/loaders/6c338d109dba8545edae479e7e2c793d.lua"))()
            end)
        end
    })

    TabOther:Section({ Title = "挂机防踢" })
    TabOther:Toggle({
        Title = "开启挂机防踢",
        Value = false,
        Callback = function(state)
            antiAfkRunning = state
            if state then
                antiAfkConnection = RunService.Heartbeat:Connect(function()
                    VirtualUser:CaptureController()
                    VirtualUser:Button2Down(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
                    task.wait(0.25)
                    VirtualUser:Button2Up(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
                end)
            else
                if antiAfkConnection then
                    antiAfkConnection:Disconnect()
                    antiAfkConnection = nil
                end
            end
        end
    })

    Window:OnClose(function()
        antiAfkRunning = false
        if antiAfkConnection then
            antiAfkConnection:Disconnect()
            antiAfkConnection = nil
        end
    end)
end

Patriot.Callbacks.OnVerify = function(inputKey)
    local validKeys = {
        "LUQENBINGSHIGAY",
        "820819QWE"
    }
    for _, k in ipairs(validKeys) do
        if inputKey == k then return true end
    end
    return false
end

Patriot.Callbacks.OnFail = function(errMsg)
    warn("密钥验证失败: " .. tostring(errMsg))
end

Patriot.Callbacks.OnClose = function()
    print("用户关闭密钥面板，脚本不会启动")
end

Patriot:Launch()
